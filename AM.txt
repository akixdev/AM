Algorithm:
  1. Start.
  2. Define add_matrices(A, B) to compute element-wise sums A[i][j] + B[i][j] for all rows i and columns j.
  3. Define multiply_matrices(A, B): check that columns of A equal rows of B, initialize a zero result matrix of shape rows_A x cols_B, and compute each entry as sum(A[i][k] * B[k][j]).
  4. Define scalar_multiply(matrix, scalar) to multiply every entry of the matrix by scalar.
  5. Initialize example matrices A, B, and scalar = 3, then compute and print A + B, A x B, B x A, and 3 x A.

def add_matrices(A, B):
    return [[A[i][j] + B[i][j] for j in range(len(A[0]))] for i in range(len(A))]

def multiply_matrices(A, B):
    rows_A, cols_A = len(A), len(A[0])
    rows_B, cols_B = len(B), len(B[0])
    assert cols_A == rows_B, "Number of columns of A must equal number of rows of B"
    result = [[0] * cols_B for _ in range(rows_A)]
    for i in range(rows_A):
        for j in range(cols_B):
            result[i][j] = sum(A[i][k] * B[k][j] for k in range(cols_A))
    return result

def scalar_multiply(matrix, scalar):
    return [[scalar * x for x in row] for row in matrix]

A = [[2, 3], [4, 5]]
B = [[1, 0], [2, 1]]
scalar = 3

print("Matrix A            :", A)
print("Matrix B            :", B)
print("A + B                :", add_matrices(A, B))
print("A x B (A*B)          :", multiply_matrices(A, B))
print("B x A (B*A)          :", multiply_matrices(B, A), " <-- shows AB != BA")
print(f"{scalar} x A               :", scalar_multiply(A, scalar))

import numpy as np

print("A @ B (numpy):\n", An @ Bn)
print("A * B (numpy, element-wise!):\n", An * Bn)

5. Practical 3A: Large Sample Test for a Single Mean (Z-test)
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and norm from scipy.stats.
  3. Define z_test_single_mean(sample, mu0, sigma=None, alpha=0.05, alternative='two-sided').
  4. Convert sample to a NumPy array, find sample size n and sample mean x_bar, and compute sample standard deviation sigma (with ddof=1) if not supplied.
  5. Calculate the test statistic Z = (x_bar - mu0) / (sigma / sqrt(n)).
  6. Compute the p_value according to the chosen tail (two-sided, greater, or less) using the standard normal CDF.
  7. Compare p_value <= alpha to decide whether to Reject H0 or Fail to Reject H0, and return the summary dictionary.
  8. Generate a simulated sample of 40 observations with mu0 = 50, run the Z-test at alpha = 0.05, and display the results and interpretation.
  9. Stop.

Code:
import numpy as np
from scipy.stats import norm

def z_test_single_mean(sample, mu0, sigma=None, alpha=0.05, alternative="two-sided"):
    sample = np.asarray(sample, dtype=float)
    n = len(sample)
    x_bar = sample.mean()
    if sigma is None:
        sigma = sample.std(ddof=1)

    z_stat = (x_bar - mu0) / (sigma / np.sqrt(n))

    if alternative == "two-sided":
        p_value = 2 * (1 - norm.cdf(abs(z_stat)))
    elif alternative == "greater":
        p_value = 1 - norm.cdf(z_stat)
    elif alternative == "less":
        p_value = norm.cdf(z_stat)
    else:
        raise ValueError("alternative must be 'two-sided', 'greater', or 'less'")

    reject = p_value <= alpha
    return {
        "n": n,
        "Sample Mean": round(x_bar, 4),
        "Hypothesized Mean": mu0,
        "Z-statistic": round(z_stat, 4),
        "p-value": round(p_value, 6),
        "alpha": alpha,
        "Conclusion": "Reject H0" if reject else "Fail to Reject H0",
    }

np.random.seed(1)
data = np.random.normal(loc=50.6, scale=2.0, size=40)
mu0 = 50

result = z_test_single_mean(data, mu0, alpha=0.05, alternative="two-sided")
for k, v in result.items():
    print(f"{k:20s}: {v}")

print("\nInterpretation: since the p-value is compared against alpha = 0.05,")
print("we", "REJECT" if result["Conclusion"] == "Reject H0" else "DO NOT REJECT",
      "the claim that the true mean fill volume is 50 ml.")

6. Practical 3D: F-test for Comparison of Variances (Small Samples)
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and f from scipy.stats.
  3. Define f_test_variances(sample1, sample2, alpha=0.05).
  4. Convert both samples to NumPy arrays and compute their sample variances var1, var2 (with ddof=1) and sample sizes n1, n2.
  5. Place the larger variance in the numerator to calculate F = max(var1, var2) / min(var1, var2) and assign the corresponding degrees of freedom (df1, df2).
  6. Compute the two-tailed p_value using the F-distribution CDF and the critical value f_dist.ppf(1 - alpha / 2, df1, df2).
  7. Compare F > critical_value to decide whether to reject or fail to reject H0, and return the result dictionary.
  8. Pass sample data for machine_A and machine_B into f_test_variances and print the output and interpretation.
  9. Stop.

Code:
import numpy as np
from scipy.stats import f as f_dist

def f_test_variances(sample1, sample2, alpha=0.05):
    s1 = np.asarray(sample1, dtype=float)
    s2 = np.asarray(sample2, dtype=float)

    var1, var2 = s1.var(ddof=1), s2.var(ddof=1)
    n1, n2 = len(s1), len(s2)

    if var1 >= var2:
        F = var1 / var2
        df1, df2 = n1 - 1, n2 - 1
    else:
        F = var2 / var1
        df1, df2 = n2 - 1, n1 - 1

    p_value = 2 * min(f_dist.cdf(F, df1, df2), 1 - f_dist.cdf(F, df1, df2))
    critical_value = f_dist.ppf(1 - alpha / 2, df1, df2)
    reject = F > critical_value

    return {
        "Variance 1": round(var1, 4),
        "Variance 2": round(var2, 4),
        "F-statistic": round(F, 4),
        "df1, df2": (df1, df2),
        "Critical F (alpha/2)": round(critical_value, 4),
        "p-value": round(p_value, 6),
        "Conclusion": "Reject H0 (variances differ)" if reject else "Fail to Reject H0 (variances equal)",
    }

machine_A = [10.2, 9.8, 10.5, 10.1, 9.9, 10.3, 10.0]
machine_B = [9.5, 10.6, 9.2, 10.8, 9.4, 10.9, 9.1]

result = f_test_variances(machine_A, machine_B, alpha=0.05)
for k, v in result.items():
    print(f"{k:22s}: {v}")

print("\nInterpretation: F compares the spread (variance) of the two machines' output.")
print("A large F far from 1, exceeding the critical value, means the machines differ")
print("significantly in consistency/precision, not just in average output.")

7. Practical 4B: n-Step Transition Probability of a Markov Chain
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and matplotlib.pyplot.
  3. Define validate_transition_matrix(P) to verify that P is square, every row sums to 1, and all entries lie in [0, 1].
  4. Define n_step_transition_probability(P, n) to validate P and compute P^n using np.linalg.matrix_power(P, n).
  5. Initialize the 3 x 3 weather transition matrix P and state names (Sunny, Cloudy, Rainy).
  6. Compute and display P^n for n = 1, 2, 5, 10, 20.
  7. Collect the first row of P^n (starting from Sunny) for n = 1 to 20 and plot how each state's probability evolves toward steady state.
  8. Stop.

Code:
import numpy as np
import matplotlib.pyplot as plt

def validate_transition_matrix(P):
    P = np.array(P, dtype=float)
    if P.ndim != 2 or P.shape[0] != P.shape[1]:
        raise ValueError("Transition matrix must be square (n x n).")
    if not np.allclose(P.sum(axis=1), 1):
        raise ValueError("Each row of the transition matrix must sum to 1.")
    if np.any(P < 0):
        raise ValueError("Transition matrix cannot have negative entries.")
    if np.any(P > 1):
        raise ValueError("Transition matrix cannot have entries greater than 1.")
    return P

def n_step_transition_probability(P, n):
    P = validate_transition_matrix(P)
    return np.linalg.matrix_power(P, n)

P = np.array([
    [0.7, 0.2, 0.1],
    [0.1, 0.6, 0.3],
    [0.4, 0.2, 0.4],
])
state_names = ["Sunny", "Cloudy", "Rainy"]

for n in [1, 2, 5, 10, 20]:
    Pn = n_step_transition_probability(P, n)
    print(f"\n{n}-step transition matrix P^{n}:\n{np.round(Pn, 4)}")

print("\nExample reading: P^5[0][2] = probability of Rainy in 5 steps, given it is Sunny today.")

max_n = 20
evolution = np.array([n_step_transition_probability(P, n)[0] for n in range(1, max_n + 1)])

plt.figure(figsize=(8, 5))
for i, name in enumerate(state_names):
    plt.plot(range(1, max_n + 1), evolution[:, i], marker="o", label=f"P(Sunny -> {name})")
plt.xlabel("Number of steps (n)")
plt.ylabel("Transition probability")
plt.title("n-step Transition Probability starting from 'Sunny'")
plt.legend()
plt.grid(alpha=0.3)
plt.show()

print("\nNote: as n grows large, every row of P^n converges to the same values —")
print("this limiting row is the Markov chain's steady-state (stationary) distribution.")

8. Practical 4D: Simulating a Higher-Order (Second-Order) Markov Process
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import defaultdict, Counter from collections, and numpy.
  3. Store the historical sequence of weather observations in a list weather.
  4. Iterate from index 2 to the end of weather, form each pair of previous two states (weather[i-2], weather[i-1]), and count occurrences of the next state weather[i].
  5. For each state pair, divide each next-state count by the total transitions from that pair to obtain conditional transition probabilities.
  6. Print all learned second-order transition probabilities in percentage format.
  7. Take the last two states current_pair = (weather[-2], weather[-1]); if present in the table, sample and print the predicted next state using rng.choice(states, p=probabilities).
  8. Stop.

Code:
from collections import defaultdict, Counter
import numpy as np

weather = [
    "Sunny", "Cloudy", "Rainy", "Rainy",
    "Cloudy", "Sunny", "Cloudy", "Rainy",
    "Sunny", "Sunny", "Cloudy", "Rainy",
    "Rainy", "Cloudy", "Sunny", "Cloudy"
]

transition_counts = defaultdict(Counter)

for i in range(2, len(weather)):
    previous_two = (weather[i - 2], weather[i - 1])
    next_state = weather[i]
    transition_counts[previous_two][next_state] += 1

transition_probabilities = {}

for pair, counts in transition_counts.items():
    total = sum(counts.values())

    transition_probabilities[pair] = {
        state: count / total
        for state, count in counts.items()
    }

print("Learned transition probabilities:")

for pair, probs in transition_probabilities.items():
    print(f"\n{pair} ->")

    for state, probability in probs.items():
        print(f"  {state}: {probability:.2%}")

current_pair = (weather[-2], weather[-1])

if current_pair in transition_probabilities:
    probs = transition_probabilities[current_pair]

    states = list(probs.keys())
    probabilities = list(probs.values())

    rng = np.random.default_rng(42)
    prediction = rng.choice(states, p=probabilities)

    print("\nPrevious two states:", current_pair)
    print("Predicted next state:", prediction)
else:
    print("Insufficient data for this pair.")

9. Practical 5A: Karl Pearson's Coefficient of Correlation
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and matplotlib.pyplot.
  3. Define pearson_correlation(x, y): convert inputs to NumPy arrays, compute means x_mean and y_mean, calculate numerator sum((x - x_mean) * (y - y_mean)) and denominator sqrt(sum((x - x_mean)^2)) * sqrt(sum((y - y_mean)^2)), and return their ratio.
  4. Initialize sample lists hours_studied and marks_obtained.
  5. Compute the correlation manually with pearson_correlation and verify using np.corrcoef.
  6. Classify the correlation strength (strong/moderate positive, weak/no linear, or moderate/strong negative) using conditional thresholds and print the interpretation.
  7. Plot a scatter diagram of the data points overlaid with the linear best-fit line obtained via np.polyfit.
  8. Stop.

Code:
import numpy as np
import matplotlib.pyplot as plt
def pearson_correlation(x, y):
    x = np.asarray(x, dtype=float)
    y = np.asarray(y, dtype=float)
    x_mean, y_mean = x.mean(), y.mean()

    numerator = np.sum((x - x_mean) * (y - y_mean))
    denominator = np.sqrt(np.sum((x - x_mean) ** 2)) * np.sqrt(np.sum((y - y_mean) ** 2))
    return numerator / denominator

hours_studied = [2, 4, 6, 8, 10, 3, 7, 5, 9, 1]
marks_obtained = [50, 60, 68, 82, 90, 55, 75, 65, 88, 40]

r_manual = pearson_correlation(hours_studied, marks_obtained)
r_numpy = np.corrcoef(hours_studied, marks_obtained)[0, 1]

print(f"Pearson correlation coefficient (from scratch) : {r_manual:.4f}")
print(f"Pearson correlation coefficient (numpy check)  : {r_numpy:.4f}")

if r_manual > 0.7:
    strength = "strong positive"
elif r_manual > 0.3:
    strength = "moderate positive"
elif r_manual > -0.3:
    strength = "weak/no linear"
elif r_manual > -0.7:
    strength = "moderate negative"
else:
    strength = "strong negative"
print(f"Interpretation: a {strength} linear relationship between hours studied and marks.")

plt.figure(figsize=(7, 5))
plt.scatter(hours_studied, marks_obtained, color="teal", s=60, label="Data points")
m, c = np.polyfit(hours_studied, marks_obtained, 1)
xs = np.linspace(min(hours_studied), max(hours_studied), 100)
plt.plot(xs, m * xs + c, "r--", label=f"Best-fit line (slope={m:.2f})")
plt.title(f"Hours Studied vs Marks Obtained  (r = {r_manual:.3f})")
plt.xlabel("Hours Studied")
plt.ylabel("Marks Obtained")
plt.legend()
plt.grid(alpha=0.3)
plt.show()

11. Practical 6A: Lagrange's Interpolation Method
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and matplotlib.pyplot.
  3. Define lagrange_interpolation(x_points, y_points, x_val): initialize result = 0.0; for each data point i, set term = y_points[i], multiply by (x_val - x_points[j]) / (x_points[i] - x_points[j]) for all j != i, and add term to result.
  4. Define known data arrays x and y and target point x_new = 2.5.
  5. Compute y_new = lagrange_interpolation(x, y, x_new) and print the interpolated value.
  6. Generate 200 evenly spaced points across [min(x), max(x)], evaluate the Lagrange polynomial at each point, and plot the curve with known and interpolated points marked.
  7. Stop.

Code:
import numpy as np
import matplotlib.pyplot as plt

def lagrange_interpolation(x_points, y_points, x_val):
    n = len(x_points)
    result = 0.0
    for i in range(n):
        term = y_points[i]
        for j in range(n):
            if i != j:
                term *= (x_val - x_points[j]) / (x_points[i] - x_points[j])
        result += term
    return result

x = np.array([1, 2, 3, 4, 5])
y = np.array([2, 4, 1, 3, 7])
x_new = 2.5

y_new = lagrange_interpolation(x, y, x_new)
print(f"Interpolated value at x = {x_new} is y = {y_new:.4f}")

x_range = np.linspace(min(x), max(x), 200)
y_range = [lagrange_interpolation(x, y, xi) for xi in x_range]

plt.figure(figsize=(8, 5))
plt.scatter(x, y, color="red", zorder=5, label="Known data points")
plt.plot(x_range, y_range, color="blue", label="Lagrange interpolation curve")
plt.scatter([x_new], [y_new], color="green", zorder=5, marker="x", s=100,
            label=f"Interpolated point (x={x_new})")
plt.title("Lagrange Interpolation")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.grid(alpha=0.3)
plt.show()

12. Practical 6B: Newton's (Divided-Difference) Interpolation Method
------------------------------------------------------------------------

Algorithm:
  1. Start.
  2. Import numpy and matplotlib.pyplot.
  3. Define divided_diff(x, y): initialize an n x n zero matrix table, fill column 0 with y, and compute higher-order divided differences table[i][j] = (table[i+1][j-1] - table[i][j-1]) / (x[i+j] - x[i]).
  4. Define newton_interpolation(x, y, value): build the divided-difference table, initialize result = table[0][0] and product_term = 1.0, and iteratively update product_term *= (value - x[i-1]) and result += table[0][i] * product_term.
  5. Initialize data points x_points = [1, 2, 3, 4] and y_points = [1, 4, 9, 16], compute and print the divided-difference table.
  6. Evaluate the interpolated value at x = 2.5 using newton_interpolation and print the result.
  7. Generate 200 evenly spaced points across [min(x_points), max(x_points)], compute the interpolating curve, and plot it with the known and interpolated points.
  8. Stop.

Code:
import numpy as np
import matplotlib.pyplot as plt

def divided_diff(x, y):
    n = len(y)
    table = [[0.0] * n for _ in range(n)]
    for i in range(n):
        table[i][0] = y[i]
    for j in range(1, n):
        for i in range(n - j):
            table[i][j] = (table[i + 1][j - 1] - table[i][j - 1]) / (x[i + j] - x[i])
    return table

print(table)

def newton_interpolation(x, y, value):
    n = len(x)
    table = divided_diff(x, y)
    result = table[0][0]
    product_term = 1.0
    for i in range(1, n):
        product_term *= (value - x[i - 1])
        result += table[0][i] * product_term
    return result

x_points = [1, 2, 3, 4]
y_points = [1, 4, 9, 16]

table = divided_diff(x_points, y_points)
print("Divided-difference table (row i, column j = f[x_i,...,x_{i+j}]):")
for row in table:
    print(["%.4f" % v for v in row])

value_to_interpolate = 2.5
interpolated_value = newton_interpolation(x_points, y_points, value_to_interpolate)
print(f"\nInterpolated value at x = {value_to_interpolate} is {interpolated_value:.4f}")
print("(Exact x^2 value would be 6.25 - matches, since the data is exactly quadratic.)")

x_range = np.linspace(min(x_points), max(x_points), 200)
y_range = [newton_interpolation(x_points, y_points, xi) for xi in x_range]

plt.figure(figsize=(8, 5))
plt.scatter(x_points, y_points, color="red", zorder=5, label="Known data points")
plt.plot(x_range, y_range, color="purple", label="Newton interpolation curve")
plt.scatter([value_to_interpolate], [interpolated_value], color="green", marker="x",
            s=100, zorder=5, label=f"Interpolated point (x={value_to_interpolate})")
plt.title("Newton's Divided-Difference Interpolation")
plt.xlabel("x")
plt.ylabel("y")
plt.legend()
plt.grid(alpha=0.3)
plt.show()
